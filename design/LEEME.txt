Para regenerar las versiones optimizadas del hero:

  const sharp=require('sharp');
  sharp('design/foto1.png').resize({width:1600}).webp({quality:72}).toFile('public/hero-1600.webp');
  sharp('design/foto1.png').resize({width:800}).webp({quality:72}).toFile('public/hero-800.webp');
  sharp('design/foto1.png').resize({width:1600}).jpeg({quality:78,mozjpeg:true}).toFile('public/hero-1600.jpg');
  "
